function tzOffset(t: number, tz: string) {
  const p: any = {};
  for (const x of new Intl.DateTimeFormat("en-US", { timeZone: tz, hourCycle: "h23", year: "numeric", month: "2-digit", day: "2-digit", hour: "2-digit", minute: "2-digit", second: "2-digit" }).formatToParts(new Date(t))) p[x.type] = x.value;
  return Date.UTC(+p.year, +p.month - 1, +p.day, +p.hour, +p.minute, +p.second) - t;
}
function ukToUtc(date: string, time: string) {
  const [d, m, y] = date.split("/").map(Number);
  const yy = y < 100 ? 2000 + y : y;
  const [hh, mm] = (time || "15:00").split(":").map(Number);
  const guess = Date.UTC(yy, m - 1, d, hh || 0, mm || 0);
  return guess - tzOffset(guess, "Europe/London");
}

function parseCsv(txt: string) {
  const lines = txt.replace(/^﻿/, "").split(/\r?\n/).filter(l => l.trim());
  const head = lines[0].split(",").map(s => s.trim().replace(/^﻿/, ""));
  return lines.slice(1).map(l => { const row = l.split(","); const o: any = {}; head.forEach((h, i) => (o[h] = row[i])); return o; });
}

  const out: M[] = []; const curTeams = new Set<string>(); const allTeams = new Set<string>();

    for (const o of parseCsv(await r.text())) {

      const t = ukToUtc(o.Date, o.Time);
      if (t < from) continue;
      allTeams.add(o.Home); allTeams.add(o.Away);
      if (t > cur) { curTeams.add(o.Home); curTeams.add(o.Away); }
      const hg = num(o, "HG"), ag = num(o, "AG");
      if (isNaN(hg) || isNaN(ag)) continue;
      out.push({ div, dk: dayKey(o.Date), t, h: o.Home, a: o.Away, hg, ag });
    }
    return { ms: out.sort((x, y) => x.t - y.t), curTeams, allTeams };
  }
  for (const s of SEASONS) {
    const r = await fetch(`https://www.football-data.co.uk/mmz4281/${s}/${div}.csv`);
    if (!r.ok) { console.log("history skip", div, s, r.status); continue; }
    for (const o of parseCsv(await r.text())) {
      if (!o.HomeTeam || !o.Date) continue;
      allTeams.add(o.HomeTeam); allTeams.add(o.AwayTeam);
      if (s === SEASONS[SEASONS.length - 1]) { curTeams.add(o.HomeTeam); curTeams.add(o.AwayTeam); }

function norm(s: string) {
  return s.normalize("NFD").replace(/[̀-ͯ]/g, "").toLowerCase().replace(/&/g, " ").replace(/[^a-z0-9 ]/g, " ").split(/\s+/).filter(w => w && !STOP.has(w)).join(" ");
}
function sim(a: string, b: string) {
  if (!a || !b) return 0;
  if (a === b) return 1;
  if (a.includes(b) || b.includes(a)) return 0.85;
  const ta = new Set(a.split(" ")), tb = new Set(b.split(" "));
  const inter = [...ta].filter(x => tb.has(x) || [...tb].some(y => y.length > 3 && x.length > 3 && (x.startsWith(y) || y.startsWith(x)))).length;
  return inter / Math.max(ta.size, tb.size);
}
const mapCache: Record<string, string | null> = {};
function mapTeam(div: string, team: any, teams: Set<string>, minScore = 0.5) {
  const key = `${div}|${team.id}|${minScore}`;
  if (key in mapCache) return mapCache[key];
  let best: string | null = null, bestS = 0;
  for (const n of [team.name, team.shortName].filter(Boolean)) {
    const nn = norm(n);
    if (ALIAS[nn] && teams.has(ALIAS[nn])) { best = ALIAS[nn]; bestS = 1; break; }
    for (const c of teams) { const s = sim(nn, norm(c)); if (s > bestS) { bestS = s; best = c; } }
  }
  const res = bestS >= minScore ? best : null;
  if (!res && minScore === 0.5) console.log("UNMAPPED", div, team.name, team.shortName, "best", best, bestS.toFixed(2));
  mapCache[key] = res;
  return res;
}

function fit(train: M[], now: number, clWeight = 1) {

  const teams = new Set<string>(); train.forEach(m => { teams.add(m.h); teams.add(m.a); });
  const A: any = {}, D: any = {}, N: any = {};
  teams.forEach(t => { A[t] = 1; D[t] = 1; N[t] = 0; });
  const w = train.map(m => Math.exp(-lam * (now - m.t)) * (m.cl ? clWeight : 1));
  train.forEach((m, i) => { N[m.h] += w[i]; N[m.a] += w[i]; });
  let H = 1.3;
  for (let it = 0; it < 40; it++) {
    const sn: any = {}, sd: any = {}; teams.forEach(t => { sn[t] = 0; sd[t] = 0; });
    train.forEach((m, i) => { sn[m.h] += w[i] * m.hg; sd[m.h] += w[i] * D[m.a] * H; sn[m.a] += w[i] * m.ag; sd[m.a] += w[i] * D[m.h]; });
    teams.forEach(t => (A[t] = (sn[t] + 1) / (sd[t] + 1)));
    const mean = [...teams].reduce((s, t) => s + A[t], 0) / teams.size; teams.forEach(t => (A[t] /= mean));
    const cn: any = {}, cd: any = {}; teams.forEach(t => { cn[t] = 0; cd[t] = 0; });
    train.forEach((m, i) => { cn[m.a] += w[i] * m.hg; cd[m.a] += w[i] * A[m.h] * H; cn[m.h] += w[i] * m.ag; cd[m.h] += w[i] * A[m.a]; });
    teams.forEach(t => (D[t] = (cn[t] + 1) / (cd[t] + 1)));
    let hn = 0, hd = 0; train.forEach((m, i) => { hn += w[i] * m.hg; hd += w[i] * A[m.h] * D[m.a]; }); H = hn / hd;
  }
  return { A, D, H, N };
}
function pois(k: number, l: number) { let p = Math.exp(-l); for (let i = 1; i <= k; i++) p *= l / i; return p; }
function tau(x: number, y: number, l: number, mu: number) {
  if (x === 0 && y === 0) return 1 - l * mu * RHO; if (x === 0 && y === 1) return 1 + l * RHO;
  if (x === 1 && y === 0) return 1 + mu * RHO; if (x === 1 && y === 1) return 1 - RHO; return 1;
}
function predict(l: number, mu: number) {
  let ph = 0, pd = 0, pa = 0, o25 = 0, best = 0, bs = "0-0", tot = 0;
  for (let x = 0; x <= MAXG; x++) for (let y = 0; y <= MAXG; y++) {
    const p = pois(x, l) * pois(y, mu) * tau(x, y, l, mu); tot += p;
    if (x > y) ph += p; else if (x === y) pd += p; else pa += p;
    if (x + y > 2) o25 += p; if (p > best) { best = p; bs = `${x}-${y}`; }
  }
  return { ph: ph / tot, pd: pd / tot, pa: pa / tot, o25: o25 / tot, score: bs, scoreP: best / tot };
}
function form(ms: M[], team: string, before: number, n = 5) {
  const g = ms.filter(m => m.t < before && (m.h === team || m.a === team)).slice(-n);
  const s = g.map(m => { const f = m.h === team ? m.hg : m.ag, a = m.h === team ? m.ag : m.hg; return f > a ? "W" : f === a ? "D" : "L"; }).join("");
  const gf = g.reduce((s, m) => s + (m.h === team ? m.hg : m.ag), 0), ga = g.reduce((s, m) => s + (m.h === team ? m.ag : m.hg), 0);
  return { s, gf, ga };
}
function venueRecord(ms: M[], team: string, home: boolean, before: number, n = 6) {

}

}
